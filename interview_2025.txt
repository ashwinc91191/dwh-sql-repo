-- 1. Employees earning > 9000 with department
SELECT e.first_name, e.last_name, d.department_name
FROM employees e
JOIN departments d ON e.department_id = d.department_id
WHERE e.salary > 9000;

-- 2. Departments having more than 3 employees
SELECT department_id, COUNT(*) AS emp_count
FROM employees
GROUP BY department_id
HAVING COUNT(*) > 3;

-- 3. Employees hired after 2020-01-01
SELECT first_name, hire_date
FROM employees
WHERE hire_date > '2020-01-01';

-- 4. Names containing 'an' ordered by last name
SELECT first_name, last_name
FROM employees
WHERE first_name LIKE '%an%'
ORDER BY last_name;

-- 5. Salary difference from company average
SELECT first_name, salary,
       salary - (SELECT AVG(salary) FROM employees) AS diff_from_avg
FROM employees;

-- 6. Employee count per location
SELECT l.location_id, l.city, COUNT(e.emp_id) AS emp_count
FROM employees e
JOIN departments d ON e.department_id = d.department_id
JOIN locations l ON d.location_id = l.location_id
GROUP BY l.location_id, l.city;

-- 7. Count distinct job roles
SELECT COUNT(DISTINCT job_id) AS distinct_jobs
FROM employees;

-- 8. Categorize employees by salary
SELECT first_name, salary,
       CASE
         WHEN salary > 10000 THEN 'High'
         WHEN salary BETWEEN 5000 AND 10000 THEN 'Medium'
         ELSE 'Low'
       END AS salary_category
FROM employees;

-- 9. Full name column
SELECT CONCAT(first_name, ' ', last_name) AS full_name FROM employees;

-- 10. Average salary per department (2 decimals)
SELECT department_id, ROUND(AVG(salary), 2) AS avg_salary
FROM employees
GROUP BY department_id;

-- 11. Employees working in location 1700
SELECT first_name, last_name
FROM employees
WHERE department_id IN (
    SELECT department_id FROM departments WHERE location_id = 1700
);

-- 12. Sort by dept, salary DESC
SELECT * FROM employees ORDER BY department_id, salary DESC;

-- 13. Departments without employees
SELECT d.department_id, d.department_name
FROM departments d
LEFT JOIN employees e ON d.department_id = e.department_id
WHERE e.emp_id IS NULL;

-- 14. UNION example
SELECT job_id FROM employees
UNION
SELECT job_id FROM jobs;

-- 15. Top 5 salaries
SELECT first_name, salary FROM employees ORDER BY salary DESC LIMIT 5;

-- 16. Count high/low earners
SELECT
  SUM(CASE WHEN salary > 10000 THEN 1 ELSE 0 END) AS high_earners,
  SUM(CASE WHEN salary <= 10000 THEN 1 ELSE 0 END) AS low_earners
FROM employees;

-- 17. Departments ordered by average salary
SELECT department_id, AVG(salary) AS avg_salary
FROM employees
GROUP BY department_id
ORDER BY avg_salary DESC;

-- 18. Three-table join
SELECT e.first_name, j.job_title, d.department_name
FROM employees e
JOIN jobs j ON e.job_id = j.job_id
JOIN departments d ON e.department_id = d.department_id;

-- 19. Employee with manager name
SELECT e.first_name AS employee, m.first_name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.emp_id;

-- 20. List employees per department
SELECT d.department_name,
       GROUP_CONCAT(CONCAT(e.first_name, ' ', e.last_name) SEPARATOR ', ') AS employees
FROM employees e
JOIN departments d ON e.department_id = d.department_id
GROUP BY d.department_name;


--Advanced

-- 1. Rank employees by salary
SELECT emp_id, first_name, salary,
       DENSE_RANK() OVER (ORDER BY salary DESC) AS salary_rank
FROM employees;

-- 2. Top 3 earners per department
SELECT * FROM (
  SELECT emp_id, first_name, department_id, salary,
         ROW_NUMBER() OVER (PARTITION BY department_id ORDER BY salary DESC) AS rn
  FROM employees
) ranked
WHERE rn <= 3;

-- 3. Total salary per department using CTE
WITH dept_total AS (
  SELECT department_id, SUM(salary) AS total_salary
  FROM employees
  GROUP BY department_id
)
SELECT * FROM dept_total ORDER BY total_salary DESC;

-- 4. Recursive CTE – employee hierarchy
WITH RECURSIVE emp_hierarchy AS (
  SELECT emp_id, first_name, manager_id, 1 AS level
  FROM employees
  WHERE manager_id IS NULL
  UNION ALL
  SELECT e.emp_id, e.first_name, e.manager_id, h.level + 1
  FROM employees e
  JOIN emp_hierarchy h ON e.manager_id = h.emp_id
)
SELECT * FROM emp_hierarchy ORDER BY level;

-- 5. Salary greater than dept average
SELECT first_name, salary, department_id
FROM employees e
WHERE salary > (
  SELECT AVG(salary)
  FROM employees
  WHERE department_id = e.department_id
);

-- 6. Count high/low earners per job
SELECT job_id,
       SUM(CASE WHEN salary > 10000 THEN 1 ELSE 0 END) AS high_earners,
       SUM(CASE WHEN salary <= 10000 THEN 1 ELSE 0 END) AS low_earners
FROM employees
GROUP BY job_id;

-- 7. Salary difference from department average
SELECT emp_id, first_name, salary,
       AVG(salary) OVER (PARTITION BY department_id) AS dept_avg,
       salary - AVG(salary) OVER (PARTITION BY department_id) AS diff_from_dept_avg
FROM employees;

-- 8. Salary range per department
SELECT department_id,
       MAX(salary) - MIN(salary) AS salary_range
FROM employees
GROUP BY department_id;

-- 9. Departments above company average
WITH company_avg AS (
  SELECT AVG(salary) AS avg_sal FROM employees
)
SELECT department_id, AVG(salary) AS dept_avg
FROM employees, company_avg
GROUP BY department_id
HAVING dept_avg > company_avg.avg_sal;

-- 10. Hires per year
SELECT YEAR(hire_date) AS hire_year, COUNT(*) AS hires
FROM employees
GROUP BY YEAR(hire_date)
ORDER BY hire_year;
